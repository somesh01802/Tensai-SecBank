    \"incident.h2_title\" AS h2t, \"incident.h2_confidence\" AS h2c, \"incident.h2_detail\" AS h2d,
    \"incident.h3_title\" AS h3t, \"incident.h3_confidence\" AS h3c, \"incident.h3_detail\" AS h3d
    FROM \"bank-problems\" WHERE \"incident.posted_at\" IS NOT NULL AND \"labels.data_source\" = 'detector' ORDER BY \"incident.posted_at\" DESC LIMIT 1"
